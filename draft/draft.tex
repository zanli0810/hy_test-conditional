\begin{section}{Hypothesis Test}

We note that that the score function of IPW is
\begin{equation*}\label{psi_W}
    \psi_W (\btheta; \x, y, \delta) = \frac{\delta}{\pi(\x,y)} \nabla_{\btheta} \dl  (y\mid \x,\btheta),
\end{equation*}
and 
\begin{align}\label{eq:score_W}
    \sum_{i=1}^{N} \psi_W (\htheta_W; \x_i, y_i, \delta_i) = 0,
\end{align}
where $\htheta_W$ is the IPW estimator.
Samewise, the score function of MSCLE is 
\begin{align}
    \psi_S (\btheta; \x, y, \delta) &= \delta [\nabla_{\btheta} \dl (y\mid \x,\btheta) - \log \bapi(\x, y)],
\end{align}
and
\begin{align}\label{eq:score_S}
    \sum_{i=1}^{N} \psi_S (\htheta_S; \x_i, y_i, \delta_i) = 0,
\end{align} 
 where $\htheta_S$ is the MSCLE and $\bapi(\x, y) = \int f(y \mid \x,\btheta) \pi(\x, y)$.
As it is mentioned by \cite{wang2022maximum}, the IPW estimator $\hbeta_W$ is
consistent, because the conditional expectation of the selection indicator
$\delta$ give the data equals the inclusion probability and therefore the
expectation of the objective function for the IPW estimator is the same as that
of the full data MLE.
Yet, the MSCLE is estimated by the assumed model structure to derive the subsampled conditional likelihood, 
so if the subsampling probabilities also depend on the data then the MSCLE is not consistent to the same limit as the full data MLE when the assumed model structure is misspecified. 
That brings a thought that the distance between IPW estimator and MSCLE can be used as an indicator of model misspecification since this distance vanishes asymptotically in the absence of misspecification, but generally doesn't vanish otherwise.
Hausman-type test \cite{hausman1978specification, white1982maximum} is established to evaluate the difference between these two parameters, i.e., one is stable when no misspecification, while the othe is more efficient yet not that stable.
The misspecification indicator is $\htheta_S - \htheta_W$, so we investigate the asymptotic distribution of $\sqrt{N} (\htheta_S - \htheta_W)$. 
We anticipate that with appropriate regularity conditions, $\sqrt{N} (\htheta_S - \htheta_W)$ will be normally distributed asymptotically with mean zero in the absence of misspecification. 
Given a consistent estimator for the asymptotic covariance matrix, we can form an asymptotic chi-square statistic. 
\begin{assumption}
    \begin{itemize}
        \item[(1)] $\| \psi_{\mathcal{T}} (\btheta; \x, y, \delta)\|, \|\nabla_{\btheta} \psi_{\mathcal{T}} (\btheta; \x, y, \delta)\|$ are integrable, where $\mathcal{T} = \{W,$ $SM\}$ and $\|A\| = tr^{1/2} (A^\top A)$ is the norm for a vector or matrix $A$.
        \item[(2)] The parameter space $\btheta$ is compact and the third order partial derivative of $\psi_{\mathcal{T}} (y\mid \x,\btheta)$ and $\log \bapi(\x, y)$ with respect to any components of $\btheta$ is bounded in absolute value by an integrable function $B(\x; y)$ that does not depend on $\btheta$. Moreover, $\Exp[B^2(\x; y)] < \infty$.
        \item[(3)] $\hat{A}_{\mathcal{T}}$ is a nonsingular matrix and the variance matrix $\Sigma_W^{-1}$ and $\Sigma_S^{-1}$ are finite and positive definite.
        \item[(4)] $\btheta_0$ is the unique solution to the population minimization problem $\min\limits_{\btheta \in \Theta} \Exp [\dl (y\mid \x,\btheta)]$.
        \item[(5)] $\Exp[\nabla_{\btheta} \psi_{\mathcal{T}} (\btheta; \x, y, \delta)]$ is a finite and positive finite matrix.
        \item[(6)] $\Sigma_W^{-1} - \Sigma_S^{-1}$ is positive semidefinite. {\color{red} However, according to the Theorem 2 in \cite{wang2022maximum}, $\Sigma_W^{-1} \geq \Sigma_S^{-1}$, hence this condition is automatically satisfied. }
        
    \end{itemize}
\end{assumption}
Besides, we also make an Assumption 2 that all the properties in Assumption 1 are satisfied for the misspecificated model $f^w$.

Assumptions~(1)--(3) provide the essential regularity conditions for the asymptotic properties of the MSCLE estimator. Assumptions~(3)--(5) furnish the necessary regularity conditions for the asymptotic properties of the IPW estimator. Assumption~(6) ensures the existence of the covariance matrix of $\htheta_S - \htheta_W$, which supports Lemma~\ref{le:test stat}.

\begin{lemma}\label{le:test stat}
    Under the null hypothesis of no misspecification, we consider the test statistic
    \begin{equation*}
        H_N = N (\htheta_S - \htheta_W) \hV_H^{-1} (\htheta_S - \htheta_W)^\top.
    \end{equation*}
    We have $\lN H_N \xrightarrow{d} \chi^2_p$,
    where $p$ is the parameter dimension and $\hV_H$ is the approximated estimation of covariate matrix of $\sqrt{N}(\htheta_S - \htheta_W)$ that $\lN \hV_H = \Sigma_W^{-1} - \Sigma_S^{-1}$. $\Sigma_W$ and $\Sigma_S$ are the inverse of asymptotic matrices of IPW estimator and MSCLE respectively.
\end{lemma}


\begin{proof}
Since $\htheta_W$ and $\htheta_S$ all have the asymptotic properties, then under the null hypothesis, it follows from \cite{liu2016inverse, wang2022maximum} that
\begin{equation*}
   \lN \sqrt{N} \htheta_W \xrightarrow{d} \N(\btheta_0, \Sigma_W^{-1}),
\end{equation*} 
and 
\begin{equation*}
   \lN \sqrt{N} \htheta_S \xrightarrow{d} \N(\btheta_0, \Sigma_S^{-1}),
\end{equation*}
where $\btheta_0$ is the true parameter.
% While under the situation of misspecification, two estimators also converge but not converge to the same parameter.
% IPW estimator still converges to the MLE of full data.
% Yet MSCLE is optimized through the expectation of prior distribution $\bapi(\x, y)$, which is related with the model assumption. 
% When the model is misspecificated, a wrong $\bapi(\x, y)$ will lead to a biased MSCLE. 

In this case, we derictly use the Hausman's Lemma
Consider the difference estimator $D_N = \htheta_S - \htheta_W$, 
we first have $\lN \sqrt{N} D_N \xrightarrow{d} \N (0, V_H)$, that the approximation estimate of $V_H$ can be derived as
\begin{equation*}
    \hV_H = \Var(\sqrt{N} (\htheta_S - \htheta_W)) \xrightarrow{d} \Sigma_W^{-1} + \Sigma_S^{-1} - 2N \Cov(\htheta_W, \htheta_S).
\end{equation*}
Then we just need to calculate the limit form of $\Cov(\htheta_W, \htheta_S)$.

$\htheta_W$ and $\htheta_S$ are the solutions of  \eqref{eq:score_W} and \eqref{eq:score_S}, so we expense these two equations at the true value $\btheta_0$ and have
\begin{align}\label{eq:tay_psi}
    0 &= \frac{1}{\sqrt{N}} \sum_{i=1}^{N} \psi_W (\btheta_0; \x_i, y_i, \delta_i) + \left[\frac{1}{N} \sum_{i=1}^{N} \nabla_{\btheta} \psi_W (\btheta_0; \x_i, y_i, \delta_i) \right] \sqrt{N} (\htheta_W - \btheta_0) + \op, \\
    0 &= \frac{1}{\sqrt{N}} \sum_{i=1}^{N} \psi_S (\btheta_0; \x_i, y_i, \delta_i) + \left[\frac{1}{N} \sum_{i=1}^{N} \nabla_{\btheta} \psi_S (\btheta_0; \x_i, y_i, \delta_i) \right] \sqrt{N} (\htheta_S - \btheta_0) + \op. 
\end{align}


Since $\Exp[\delta \mid \x,y] = \pi(\x,y)$,
\begin{align}\label{B_WS}
    \notag
    B_{WS} &= \Exp_{(\x,y)} \left[ \psi_W (\btheta_0; \x, y, \delta) \psi_S (\btheta_0; \x, y, \delta)^\top \right] \\ \notag 
    &= \Exp_{(\x,y)} \left[ \nabla_{\btheta} \dl (y\mid \x,\btheta_0) \left( \nabla_{\btheta} \dl (y\mid \x,\btheta_0) - \nabla_{\btheta} \log \bapi(\x, y)  \right)^\top \right] \\ \notag
    &= \Exp_{(\x,y)} \left[ \nabla_{\btheta} \dl (y\mid \x,\btheta_0) \nabla_{\btheta} \dl (y\mid \x,\btheta_0)^\top \right] - \Exp_{(\x,y)} \left[ \nabla_{\btheta} \dl (y\mid \x,\btheta_0) \nabla_{\btheta} \log \bapi(\x, y)^\top \right] \\ \notag
    &= \Exp_{(\x,y)} \left[ \nabla_{\btheta} \dl (y\mid \x,\btheta_0) \nabla_{\btheta} \dl (y\mid \x,\btheta_0)^\top \right] - \Exp_{(x)} \left[ \Exp_{(y \mid \x)} \left[ \nabla_{\btheta} \dl (y\mid \x,\btheta_0) \right] \nabla_{\btheta} \log \bapi(\x, y)^\top  \right] \\
    &= \Exp_{(\x,y)} \left[ \nabla_{\btheta} \dl (y\mid \x,\btheta_0) \nabla_{\btheta} \dl (y\mid \x,\btheta_0)^\top \right],
\end{align}
where we use $\Exp_{(y \mid \x)} \left[ \nabla_{\btheta} \dl (y\mid \x,\btheta_0) \right]=0$ in this case.

Besides, with the Assumption 1(1) and (5), and Eq.~\eqref{B_WS}, we have
\begin{equation*}
    A_W = - \Exp [-\nabla_{\btheta} \psi_W (\btheta_0; \x, y, \delta)] = - \Exp_{(\x,y)} \left[\nabla_{\btheta}^2 \ell_W (\btheta_0; \x, y, \delta) \right] = -B_{WS},
\end{equation*} 
while
\begin{equation*}
    A_S  = - \Exp [-\nabla_{\btheta}^2 \ell_S(\btheta_0; \x, y, \delta)] = - \Sigma_S,
\end{equation*}
with the conclusion in \cite{wang2022maximum} (A.12) that $-\frac{1}{N} \nabla_{\btheta}^2 \ell_S(\btheta_0; \x, y, \delta) = \Sigma_S + \op$.

Rearranging Eq.~\eqref{eq:tay_psi} and noting that 
$N^{-1}\sum_{i=1}^N \nabla_{\btheta}\psi_W(\btheta_0; ; \x_i, y_i, \delta_i) \cvp A_W$ with $A_W$ nonsingular (Assumption 1(5)), 
an application of Slutsky's theorem yields
\begin{equation*}
    \lN \sqrt{N} (\htheta_W - \btheta_0) = - A_W^{-1} (\frac{1}{\sqrt{N}} \sum_{i=1}^{N} \psi_W (\btheta_0; \x_i, y_i, \delta_i)) +\op,
\end{equation*}
and 
\begin{equation*}
    \lN \sqrt{N} (\htheta_S - \btheta_0) = - A_S^{-1} (\frac{1}{\sqrt{N}} \sum_{i=1}^{N} \psi_S (\btheta_0; \x_i, y_i, \delta_i)) +\op.
\end{equation*}
Then
\begin{equation*}
    \lN \Cov(\sqrt{N} \htheta_W, \sqrt{N} \htheta_S) = \lN E[(\htheta_W -\btheta_0) (\htheta_S -\btheta_0)^\top] = A_W^{-1} B_{WS} A_S^{-1 \top} = \Sigma_S^{-1}.
\end{equation*}
Recall that
\begin{equation*}
    \hV_H \cvp \Sigma_W^{-1} + \Sigma_S^{-1} - 2N \Cov(\htheta_W, \htheta_S) \cvp \Sigma_W^{-1} - \Sigma_S^{-1}.
\end{equation*}
Thus based on the property of Wald-type statistics \cite{van2000asymptotic} and Slutsky's Theorem, the test statistic $H_N$ follows a chi-square distribution $\chi_p^2$.
\end{proof}

As a result, $H_N$ can be used as a test statistic to decide whether the IPW or the MSCLE estimator should be chosen. 
% Let $\btheta_0^*$ denote the full-data maximum likelihood estimator. Note that $\htheta_W \xrightarrow{p} \btheta_0^*$ holds under any specification, whereas $\htheta_S \xrightarrow{p} \btheta_0^*$ may fail if the model is misspecified.
The corresponding hypothesis test is formulated as follows:
\begin{equation*}
    H_0: \Exp[\htheta_S - \htheta_W]=0 \leftrightarrow H_1:  \Exp[\htheta_S - \htheta_W]\neq 0,
\end{equation*}
that null hypothesis $H_0$ represents that the model assumption $\dl (y\mid \x,\btheta)$ is correct while the alternative hypothesis $H_1$ represents that the model is misspecified.
We then compute the $p$-value $p_H = 1 - F_{\chi_p^2}(H_N)$ and compare it with the predetermined significance level $\alpha$.  
If $p_H \leq alpha$, we reject $H_0$ and choose the IPW estimator for more stable estimation; otherwise, we accept $H_0$ and select the MSCLE estimator for greater efficiency.  
Thus, we define a new combined estimator that incorporates both the IPW and MSCLE estimators via the hypothesis-test indicator:
\begin{align}
    \htheta_N &= \I\{H_N \geq c_\alpha\} \htheta_W + \I\{H_N < c_\alpha\} \htheta_S,
\end{align}
where $c_\alpha$ is the upper confidence limit of level $\alpha$ for the chi-squared test and $\I(\cdot)$ is the indicator function.
We now study the property of the new estimator $\htheta_N$, considering two separate cases. 
The situation should be discussed seperately. 
First, if $H_0$ is accepted, when $n \rightarrow \infty$, we have $P(H_N > c_\alpha) \rightarrow \alpha$, then we have probability $\alpha$ to choose IPW estimator and probability $1-\alpha$ to choose MSCLE. 
When the model is correctly specified or both the IPW and MSCLE estimators converge to the same full data estimator $\btheta_l$, $\htheta_N$ is the unbias estimatior since it is the combination of two unbiased estimator
% $\sqrt{N} (\htheta_N - \btheta_0)= \sqrt{N} (\htheta_W - \btheta_0) \I\{H_N \geq c_\alpha\} + \sqrt{N} (\htheta_S - \btheta_0) \I\{H_N < c_\alpha\}$.
\begin{equation*}\label{eq:HNmean}
    \Exp[ \htheta_N - \btheta_0 ] = \Exp[\I\{H_N \geq c_\alpha\} (\htheta_W - \btheta_0)] + \Exp[\I\{H_N < c_\alpha\} (\htheta_S - \btheta_0)] = 0.
\end{equation*}
% \begin{align}\label{eq:HNmean}
%     \notag
%     & \Exp[(\htheta_W - \btheta_0) \I\{H_N \geq c_\alpha\}] = \Exp[\Exp[(\htheta_W - \btheta_0) \mid \htheta_W-\htheta_S] \I\{H_N \geq c_\alpha\}] \\
%     \notag
%     &= \Exp[N \Cov(\htheta_W - \btheta_0, \htheta_W-\htheta_S) \Sigma_{S} (\htheta_W-\htheta_S) \I\{H_N \geq c_\alpha\}] \\ 
%     &= (\Sigma_W^{-1}-\Sigma_S^{-1}) \Sigma_{S} \Exp[(\htheta_W-\htheta_S) \I\{N(\htheta_W-\htheta_S)V_{H}^{-1}(\htheta_W-\htheta_S)^\top \geq c_\alpha\}] = 0.
% \end{align}
Since
\begin{equation*}
   \sqrt{N} (\htheta_W-\htheta_S) \sim \N(0, \Sigma_S^{-1}).
\end{equation*}
and the sample space of $\{(\htheta_W-\htheta_S) \in \R^k: \I\{H_N \geq c_\alpha\}=1\}$ should be a exterior of ellipsoid, Eq.~\eqref{eq:HNmean} equals to 0.
Thus, we have $\htheta_N$ is an unbiased estiamtor.

\section{Model misspecificated estimator}

When the model is misspecificated, the score function of MSCLE would be a little different from Eq.~\eqref{eq:score_S} 
since we have to calculate the expection $\bapi(\x, y) = \int f(y \mid \x,\btheta) \pi(\x, y)$, which depends on the assumption of $\ell$.
Thus, we assume that the score function of MSCLE under model misspecification is 
\begin{align}
    \psi_{SM}(\btheta; \x_i, y_i, \delta_i) = \delta_i [\dl(y_i \mid \x_i,\btheta) -  \frac{E^w\{ \dl (y_i \mid \x_i,\btheta) \pi(y_i,\x_i;\btheta) \mid \x_i\}}{E^w\{ \pi(y_i,\x_i;\btheta) \mid \x_i\}}  ],
\end{align}
where $E^w$ is the expectation under the assumed working model.
Also, the MSCLE estimator is the solution of the following estimation equation \citep{wang2022maximum}
\begin{align}
    \Exp^t[\psi_{SM}(\btheta; \x, y, \delta)] = 0,
\end{align}
or
    \begin{align}
        \Exp^t \left\{ \Exp^t \left[ \dl(\btheta_; \x, y) \pi(\x,y;\btheta) \right]  -\frac{\Exp^t \left[ \pi(\x,y;\btheta) \mid \x \right]}{\Exp^w \left[ \pi(\x,y;\btheta) \mid \x \right]} \Exp^w \left[ \dl(\btheta; \x, y) \pi(\x,y;\btheta) \right] \right\} = 0,
    \end{align}
For the IPW estimator, it is the same as that of the full data MLE and be solved through the function of 
\begin{align}
    \Exp^t [\dl (\btheta; \x,y)]=0,
\end{align}
where we use $\Exp^t$ to emphasize that the expectation is taken with respect to the true data
distribution.
Since IPW and MSCLE converges to different parameter estimations, we 
assume that IPW and MSCLE converages to $\btheta_l$ and $\btheta_{ls}$ respectively. 

We first discuss the asymptotic Properties of $\htheta_W$ and $\htheta_{SM}$ under model misspecification.
We establish the consistency and asymptotic normality of both estimators using the framework of \citet{hjort2011asymptotics}.
A key advantage of this approach is that it does not require the 
assumed model to be correctly specified: the results hold with the pseudo-true values $\btheta_l$ and $\btheta_{ls}$ replacing the true parameter $\btheta_0$.
\begin{theorem}
    \begin{align}
        \sqrt{N}(\htheta_W - \btheta_l) \xrightarrow{d} \N(0, \Sigma_W^{-1}),
    \end{align}
    where $\Sigma_W^{-1} = A_W^{-1} B_W A_W^{-\top}$ with $B_W = \Exp[\psi_W(\btheta_l; \x, y,\delta) \psi_W(\btheta_l; \x, y,\delta)^\top]$, 
    and
    \begin{align}
        \sqrt{N}(\htheta_S - \btheta_{ls}) \xrightarrow{d} \N(0,  \Sigma_{SM}^{-1}),
    \end{align}
    where $\Sigma_{SM}^{-1} = A_{SM}^{-1} B_{SM} A_{SM}^{-\top}$ with $B_{SM} = \Exp[\psi_{SM}(\btheta_{ls}; \x, y,\delta) \psi_{SM}(\btheta_{ls}; \x, y,\delta)^\top]$.
    Here, $A_W = \Exp[\nabla_{\btheta} \psi_W(\btheta_l; \x, y,\delta)]$, $A_{SM} = \Exp[\nabla_{\btheta} \psi_{SM}(\btheta_{ls}; \x, y,\delta)]$.
\end{theorem}
As applied in the Theorem~1 by \citet{wang2022maximum}, the consistency and asymptotic normality of $\htheta_W$ and $\htheta_S$ follow from the Basic Corollary of \citet{hjort2011asymptotics}, which applies under possible model misspecification since the convex minimization framework does not require the model to be correctly specified.

    
Under the model misspecification, the variance-correlation matrices should also be different and we tend to present the asymptotic order of estimation of variance-correlation matrices. 
%Another difference is that $\hV_H$ has the different structure and should be more complicated. 
%In this case, we use the influence function to calculation the $\hV_H$ and calculation of variance matrices are also derived.
We assume that $A_{\mathcal{T}} = \Exp [ \nabla_{\btheta} \psi_{\mathcal{T}} (\btheta_{\mathcal{T}}; \x, y, \delta)]$, where $\mathcal{T} = \{W, SM\}$ and $\btheta_{\mathcal{T}} = \{\btheta_l, \btheta_{ls}\}$.
Let 
\begin{equation*}
    \hat{A}_{\mathcal{T}} = \frac{1}{N} \sum_{i=1}^{N} \nabla_{\btheta} \psi_{\mathcal{T}} (\htheta_{\mathcal{T}}; \x_i, y_i, \delta_i),
\end{equation*}
we first prove the following lemma.
\begin{lemma}\label{asym_AW}
    $\hat{A}_{\mathcal{T}} \cvp A_{\mathcal{T}}$.
\end{lemma}

\begin{proof}
    Let $\bar{A}_{\mathcal{T}}(\btheta) = \frac{1}{N} \sum_{i=1}^{N} \nabla_{\btheta} \psi_{\mathcal{T}} (\btheta; \x_i, y_i, \delta_i)$, then $\hat{A}_{\mathcal{T}} = \bar{A}_{\mathcal{T}}(\htheta_{\mathcal{T}})$.
    We have 
    \begin{equation*}
        \hat{A}_{\mathcal{T}} - A_{\mathcal{T}} = [\bar{A}_{\mathcal{T}}(\htheta_{\mathcal{T}}) - \bar{A}_{\mathcal{T}}(\btheta_{\mathcal{T}})] + [\bar{A}_{\mathcal{T}}(\btheta_{\mathcal{T}}) - A_{\mathcal{T}}].
    \end{equation*}
    With the Assumption 1(1), we prove that $\bar{A}_{\mathcal{T}}(\btheta_{\mathcal{T}}) \cvp A_{\mathcal{T}}$ via the law of large numbers. Next, we show that $\|\bar{A}_{\mathcal{T}}(\htheta_{\mathcal{T}}) - \bar{A}_{\mathcal{T}}(\btheta_{\mathcal{T}})\| \cvp 0$.
    To this end, we apply a Taylor expansion to each component of the derivative matrix  $\nabla_{\btheta} \psi_{\mathcal{T}} (\htheta_{\mathcal{T}}; \x_i, y_i, \delta_i)$ around $\btheta_{\mathcal{T}}$. Then for each component, we obtain
    \begin{equation*}
        [\nabla_{\btheta} \psi_{\mathcal{T}} (\htheta_{\mathcal{T}}; \x_i, y_i, \delta_i)]_{j,k} = [\nabla_{\btheta} \psi_{\mathcal{T}} (\btheta_{\mathcal{T}}; \x_i, y_i, \delta_i)]_{j,k} + [\nabla_{\btheta}^2 \psi_{\mathcal{T}} (\tilde{\btheta}_i; \x_i, y_i, \delta_i)]_{j,k} (\htheta_{\mathcal{T}} - \btheta_{\mathcal{T}}),
    \end{equation*}
    where $\tilde{\btheta}_i$ lies on the segment between 
    $\htheta_{\mathcal{T}}$ and $\btheta_{\mathcal{T}}$
    Thus,
    \begin{align*}
        \|\bar{A}_{\mathcal{T}}(\htheta_{\mathcal{T}}) - \bar{A}_{\mathcal{T}}(\btheta_{\mathcal{T}})\| &\leq \|\frac{1}{N} \sum_{i=1}^{N} \nabla_{\btheta}^2 \psi_{\mathcal{T}} (\tilde{\btheta}_i; \x_i, y_i, \delta_i)\| \|\htheta_{\mathcal{T}} - \btheta_{\mathcal{T}}\| \notag \\
        \leq \frac{1}{N} \sum_{i=1}^{N} B(\x_i, y_i) \|\htheta_{\mathcal{T}} - \btheta_{\mathcal{T}}\|.
    \end{align*}
    With the Assumption 1(2), we have $\frac{1}{N} \sum_{i=1}^{N} B(\x_i, y_i) \cvp \Exp[B(\x_i, y_i)] < \infty$ and $\|\htheta_{\mathcal{T}} - \btheta_{\mathcal{T}}\| = \op$. Then, we can prove that $\|\bar{A}_{\mathcal{T}}(\htheta_{\mathcal{T}}) - \bar{A}_{\mathcal{T}}(\btheta_{\mathcal{T}})\| \cvp 0$.
    Combining the two part above, we can finish the proof. 
\end{proof}

Besides,
we assume that
\begin{equation*}
    \hat{B}_{\mathcal{T}_1 \mathcal{T}_2} = \hat{B}_{\mathcal{T}_1 \mathcal{T}_2}\left [\frac{1}{N} \sum_{i=1}^{N} \psi_{\mathcal{T}_1} (\btheta_{\mathcal{T}_1}; \x_i, y_i, \delta_i) \right] \left[\frac{1}{N} \sum_{i=1}^{N} \psi_{\mathcal{T}_2} (\btheta_{\mathcal{T}_2}; \x_i, y_i, \delta_i) \right]^\top,
\end{equation*}
where $\mathcal{T}_1, \mathcal{T}_2$ can choose from the index set $W, SM$,
and 
\begin{equation*}
    B_{\mathcal{T}_1 \mathcal{T}_2} = \Exp [\psi_{\mathcal{T}_1}(\btheta_{\mathcal{T}_1}; \x, y, \delta) \psi_{\mathcal{T}_2} (\btheta_{\mathcal{T}_2}; \x, y, \delta)^\top].
\end{equation*}
We denote that $B_{\mathcal{T}_1 \mathcal{T}_1} = B_{\mathcal{T}_1}$ and $B_{\mathcal{T}_2 \mathcal{T}_1} = B_{\mathcal{T}_2}$ for simplify.
Then we have
\begin{lemma}\label{asym_BW}
    $\hat{B}_{\mathcal{T}_1 \mathcal{T}_2} \cvp B_{\mathcal{T}_1 \mathcal{T}_2}$.
\end{lemma}

\begin{proof}
    The proof idea is similar with the former one. We also set a middle item $\bar{B}_{\mathcal{T}_1 \mathcal{T}_2}(\htheta_1, \htheta_2) $ $= \frac{1}{N} \sum_{i=1}^{N} \psi_{\mathcal{T}_1}(\htheta_1; \x_i, y_i, \delta_i) \psi_{\mathcal{T}_2} (\htheta_2; \x_i, y_i, \delta_i)^\top$.
    So we can divide the difference of $\hat{B}_{\mathcal{T}_1 \mathcal{T}_2}$ and $B_{\mathcal{T}_1 \mathcal{T}_2}$ into two items,
    \begin{equation*}
        \hat{B}_{\mathcal{T}_1 \mathcal{T}_2} - B_{\mathcal{T}_1 \mathcal{T}_2} = [\bar{B}_{\mathcal{T}_1 \mathcal{T}_2} (\btheta_{\mathcal{T}_1}, \btheta_{\mathcal{T}_2}) - \bar{B}_{\mathcal{T}_1 \mathcal{T}_2}(\btheta_{\mathcal{T}_1}, \btheta_{\mathcal{T}_2})] + [\bar{B}_{\mathcal{T}_1 \mathcal{T}_2}(\btheta_{\mathcal{T}_1}, \btheta_{\mathcal{T}_2}) - B_{\mathcal{T}_1 \mathcal{T}_2}].
    \end{equation*}

        & \sim \op \op \Op =\op.

\end{proof}

\begin{lemma}\label{le:mis_conv}
Suppose the model is misspecified in the sense that $\btheta_l \neq \btheta_{ls}$, where the difference $\btheta_l - \btheta_{ls}$ is a fixed nonzero constant. Then, under Assumption 1,
    $H_N \xrightarrow{p} \infty, \text{and} \|\htheta_N - \htheta_W\| \cvp 0$.
\end{lemma}

\begin{proof}
Under the stated conditions, both estimators are consistent for their respective pseudo-true values:
\begin{align}
    \notag
    \sqrt{N} (\htheta_W - \btheta_l) & \xrightarrow{d} Z_W \sim N(0, \Sigma_{W}^{-1}), \\
    \sqrt{N} (\htheta_S - \btheta_{ls}) &\xrightarrow{d} Z_S \sim N(0, \Sigma_{SM}^{-1}).
\end{align}
We notice that
\begin{equation*}\label{eq:diff_order}
    \htheta_W - \htheta_S \cvp \btheta_l - \btheta_{ls} \neq 0.
\end{equation*}
Substituting~\eqref{eq:diff_order} into the test statistic and using $\hV_H \cvp V_H$,
\begin{align*}
        H_N 
        &= N(\htheta_W - \htheta_S)^\top 
        \hV_H^{-1}(\htheta_W - \htheta_S) \\
        &= N \left[(\btheta_l - \btheta_{ls}) 
        + O_P(N^{-1/2}) \right]^\top
        \left[V_H^{-1} + \op \right]
        \left[(\btheta_l - \btheta_{ls}) 
        + O_P(N^{-1/2})\right] \\
        &= N (\btheta_l - \btheta_{ls})^\top 
        V_H^{-1}(\btheta_l - \btheta_{ls}) 
        + O_P(\sqrt{N}) \rightarrow \infty.
    \end{align*} 
As a result, $\I\{H_N \geq c_\alpha\} \xrightarrow{p} 1$ for any fixed threshold $c_\alpha > 0$. 
We conclude that
\begin{equation*}
\|\htheta_{N} - \htheta_W\| \leq \underbrace{\I\{H_N < c_\alpha\}}_{\cvp\; 0}\cdot\underbrace{\|\htheta_S - \htheta_W\|}_{\Op} \cvp 0.            
\end{equation*}
\end{proof}

\subsection{A model averaging estimator}

The biggest problem of handling $\htheta_N$ is that it is a uncontinues estimator at the point of threshold $H_N = c_\alpha$. Thus, we use the model averageing method instead, by adding a smooth weight into the estimator.
The new estimator 
\begin{equation*}
    \htheta_{NA} = w(H_N)\htheta_W + (1-w(H_N))\htheta_S,
\end{equation*}
where $w(H_N)$ is a weight function related with $H_N$.
We have three kinds options to choose for $w(\cdot)$.
\begin{itemize}
    \item AIC criteria: $w(H_N) = \I\{H_N \geq c_\alpha\}$, which brings $\htheta_{NA} = \btheta_{N}$;
    \item Sigmoid Smoothing: $w(H_N) = \frac{1}{1+\exp\{-(H_N-c_\alpha)/s\}}$, where $s$ is the smoothing parameter.
    \item Adaptive weighting: $w(H_N) = \begin{cases}
 0, & \text{if} \quad H_N \leq c_\alpha, \\
1-\frac{c_\alpha}{H_N}, & \text{if} \quad H_N > c_\alpha.
\end{cases}$
\end{itemize}

\begin{assumption}
    \begin{itemize}
        \item [(1)] 
        $w(\cdot)$ is a measurable function with the value region of $[0,1]$. $w(\cdot)$ should be almost continuous function so that when $D_N \xrightarrow{d} D$, we have $w(D_N) \xrightarrow{d} w(D)$.
    \item [(2)] $A_W$ and $A_{SM}$ are nonsingular matrices.   
    \end{itemize}
\end{assumption}
For the assumption~(1), since we commonly consider $w(\cdot)$ as sigmoid or huber function, the condition is satisfied naturely.

In the Lemma~\ref{le:mis_conv}, we discussed that IPW and MSCLE may converges to different value $\btheta_l$ and $\btheta_{ls}$. 
\begin{itemize}
    \item Case 1. $\btheta_l - \btheta_{ls} = h$.
\end{itemize}
Considering the Lemma~\ref{le:mis_conv} more generally, for any weight function satisfying 
$w(h) \to 1$ as $h \to \infty$ (such as the sigmoid or the adaptive weight ), the same conclusion holds: $w(H_N) \cvp 1$ and hence 
$\|\htheta_{NA} - \htheta_W\| \cvp 0$.

\begin{itemize}
    \item Case 2. $\|\btheta_l - \btheta_{ls}\| \leq \frac{M}{\sqrt{N}}$.
\end{itemize}
We next consider that the different between $\btheta_l$ and $\btheta_{ls}$ decreases as $N$ increases. 
We assume that $\|\btheta_l - \btheta_{ls}\| \leq \frac{M}{\sqrt{N}}$, which is bounded by a value with order of $N^{-1/2}$ and $\delta_N = \sqrt{N} (\btheta_l - \btheta_{ls})$ which is a bounded sequence.
As a result, through Bolzano-Weierstrass Theorem, there exists a subsequence $\{N_{k}\}$ such that when $k \rightarrow \infty$, $\delta_{N_k}$ converages to a constant vector $\delta_s$ and $\|\delta_s\| \leq M$.
Thus let $D_N = \sqrt{N}(\htheta_W - \htheta_S)$.
 As a result, 
 \begin{align}\label{WS_Nk}

 \end{align}

\begin{equation*}

\end{equation*}
while the center value $\lambda  = \delta_s^\top V_H^{-1} \delta_s$ that can be bounded that
\begin{equation*}
    \delta_s^\top V_H^{-1} \delta_s \leq \lambda_{\text{max}}(V_H^{-1}) M^2,
\end{equation*}
where $\lambda_{\text{max}}(V_H^{-1})$ is the maximum eigenvalue of $V_H^{-1}$.

        &= tr (\Exp[ (1-w(\|D_\delta\|^2)) C_{WD} V_H^{-1} (D_\delta + \delta_s) D_\delta^\top ]),

We first consider the joint distribution of $Z_W$ and $D_\delta$, we have 
\begin{align}
    \begin{pmatrix}
        Z_W \\
        D_\delta
    \end{pmatrix}
    \sim N\left(\begin{pmatrix}
        0 \\
        -h
    \end{pmatrix},
        \begin{pmatrix}
        \Sigma_W^{-1} & C_{WD} \\
        C_{WD}^\top & V_H
    \end{pmatrix}\right),
\end{align}
where $C_{WD} = \Sigma_W^{-1} - C_{WS}$.
Then based on the conditional property of multinormal distribution, we can turn $Z_W$ to $Z_W = C_{WD} V_H^{-1} (D_\delta + h) +\epsilon$, where $\epsilon$ is a standard normal variable which is independce with $D$. 
As a result, the third item in Eq.~\eqref{eq:Rh} can be formulated as 
\begin{align}
    \notag
    &tr (\Exp(w(H_N)(Z_W(Z_S-Z_W - \delta_s)^\top)))\\
    &= tr (C_{WD}V_H^{-1} \Exp[w(\|D_\delta\|^2) (D_\delta + \delta_s) D_\delta^\top]) \notag \\
    &= tr (C_{WD}V_H^{-1} \Exp[w(\|D_\delta\|^2)] (\|D_\delta\|^2 + \delta_s D_\delta^\top)) \notag \\
    &= tr (C_{WD}V_H^{-1} \{V_H \Exp[(1 - w(\chi_{p+2}^2(\lambda)))]  + \Exp[(1 - w(\chi_{p+4}^2(\lambda)))] hh^\top \})  \notag\\
    &- tr (C_{WD}V_H^{-1} [hh^\top \Exp((1- w(\chi_{p+2}^2(\lambda))))])  \notag \\
    &= tr (C_{WD}) \Exp[(1 - w(\chi_{p+2}^2(\lambda)))] \notag \\
    &+ tr (C_{WD}V_H^{-1}) hh^\top [\Exp[(1 - w(\chi_{p+4}^2(\lambda)))^2] - E[(1 - w(\chi_{p+2}^2(\lambda)))]].
\end{align}
Thus, we complete the proof.
\end{proof}

Besides, we can also proof that when $\|\delta_s\|$ goes to infinity, the MSE of $\htheta_{NA}$ still converges to the one of IPW, showing the robustness of such estimator. 
For the second item, since $1 - w(\chi_{p+2}^2(\lambda)) \xrightarrow{p} 0$, then $V_H^{1/2} (\Exp[(1 - w(\chi_{p+2}^2(\lambda)))^2] \I_p) \rightarrow 0$. 
We also have 
    \begin{align}
        V_H^{1/2} \Exp[(1 - w(\chi_{p+4}^2(\lambda)))^2] V_H^{-1/2} \|\delta_s\|^2 V_H^{-1/2} V_H^{1/2} = \Exp[(1 - w(\chi_{p+4}^2(\lambda)))^2] \|\delta_s\|^2.
    \end{align}
To ensure that this item converges to 0 and recall that $\lambda = \delta_s^\top V_H^{-1} \delta_s^\top \sim O_{P}(\|M\|^2)$, we just need to make sure that $w(H_N) \sim O_{P}(H_N^{-1})$.
Sigmoid smoothing and adaptive weighting functions all satisfies this requirement.
For the third item, we have 
\begin{equation*}
        \lim\limits_{\|\delta_s\| \rightarrow \infty} C_{WD} \Exp[(1 - w(\chi_{p+2}^2(\lambda)))^2] \rightarrow 0,
\end{equation*}
and 
\begin{equation*}
    hh^\top [\Exp[(1 - w(\chi_{p+4}^2(\lambda)))^2] - \Exp[(1 - w(\chi_{p+2}^2(\lambda)))^2]] \rightarrow 0
\end{equation*}
when $\lambda = \delta_s^\top V_H^{-1} \delta_s \sim O_{P}(\|\delta_s\|^2)$.
Then we prove that the third item also converges to 0, so as to the fourth part.

We assume that $w^*(H_N) = \arg\min_{w} R$.
Thus, we have
\begin{lemma}\label{optimal_w_HN}
    We can obtain the optimal function $w^*$ through 
    \begin{align}\label{wstar}
        w^*(H_N) &= 1- \frac{f_{\chi_{p+2}^2(\lambda)}(H_N) [tr(C_{WD} V_H^{-1}) - tr(C_{WD})]}{f_{\chi_{p+2}^2(\lambda)}(H_N) tr(V_H) + f_{\chi_{p+4}^2(\lambda)}(H_N) \left[1+ 2 tr(C_{WD} V_H^{-1}) \right] },
    \end{align}
where $C_{WD} = \Sigma_W^{-1} - C_{WS}$.
\end{lemma}
\begin{proof}
    To minimize $R$, we differentiate $R$ in Eq.~\eqref{R_formula} with respect to $w(H_N)$, and have the following equation. 
    \begin{align}\label{R_differ}
        \notag
        & 2(1-w(H_N)) [f_{\chi_{p+2}^2(\lambda)}(H_N) tr(V_H) + f_{\chi_{p+4}^2(\lambda)}(H_N) + f_{\chi_{p+4}^2(\lambda)}(H_N) tr(C_{WD} V_H^{-1})] \\
        &= 2 f_{\chi_{p+2}^2(\lambda)}(H_N) tr(C_{WD} V_H^{-1}) - 2 tr(C_{WD}).
    \end{align}
    By solving Eq.~\eqref{R_differ}, we can obtain the optimized $w^*(H_N)$.
\end{proof}

Let $w^*(H_N)$ denote the oracle pointwise optimal weight function 
defined in Eq.~\ref{wstar}, which depends on the true asymptotic 
quantities such as $\Sigma_W$, $C_{WD}$, $V_H$, and $\lambda$. 
Since these population-level quantities are unknown in practice, 
$w^*(H_N)$ cannot be computed directly.
Instead, we replace all unknown components with their sample 
counterparts and define the plug-in weight function 
$\hat{w}^*(h_N) = w^*(h_N;  \hat{\Theta}_M)$, 
where $\hat{\Theta}_M = (\hat{\Sigma}_W, \hat{C}_{WD}, \hV_H, \hat{\lambda})$.
Accordingly, we introduce the proxy risk function $\hat{R}(\hat{w}^*)$, 
in which every population quantity appearing in $R(w)$ is replaced 
by its plug-in estimator.
Our goal is to show that the proxy risk $\hat{R}(\hat{w}^*)$ 
provides a valid approximation to the true risk $R(w^*)$ 
in the sense that their difference remains bounded as $N \to \infty$.
This strategy is motivated by the model averaging literature 
\citep{liang2011optimal, zhang2016optimal, chen2024optimal}, 
where analogous arguments are used to establish that the optimal 
weights under the estimated and true risk functions are 
asymptotically equivalent. 

However, unlike the classical model averaging framework in 
\citep{liang2011optimal, zhang2016optimal, chen2024optimal}, 
where one can establish that the ratio of the estimated risk to the oracle risk converges to one in probability, the result of $\frac{w^*(H_N)}{w^*_{or}} \cvp 1$ or $\lN \hat{R}(\hat{w}^*) \cvp R(w^*)$ does not hold in our setting.
The fundamental obstacle is that the non-centrality parameter 
$\lambda$ cannot be consistently estimated: its plug-in estimator $\hat{\lambda} = H_N$ converges in distribution to $\chi^2_p(\lambda)$, rather than in probability to the constant $\lambda$.
Instead, we establish that the excess risk 
$R(\hat{w}^*) - R(w^*)$ remains bounded by a finite constant 
that does not grow with $N$, as formalised in the following result.

\begin{proposition}
    $\lN \hat{R}(\hat{w}^*) - R(w^*) =\Op$, 
    where the bound is finite and does not vanish as $N \to \infty$.
\end{proposition}

\begin{proof}
    We recall that the risk of any weight function $w(\cdot)$ satisfies
    \begin{align*}
        R(w) &= N \Exp [\|(\htheta_W - \btheta_l) - (1-w(H_N))(\htheta_W - \htheta_S)\|^2] \notag \\
        &= N \Exp[ \Exp[\|(\htheta_W - \btheta_l) - (1-w(H_N))(\htheta_W - \htheta_S)\|^2] \mid H_N] \notag \\
        &= N \int \Exp[\|(\htheta_W - \btheta_l) - (1-w(h_N))(\htheta_W - \htheta_S)\|^2] f_{H_N}(h_N) dh_N,
    \end{align*}
    Expanding the squared norm inside the conditional expectation and noting that $w(h_N)$ is a constant given $H_N = h_N$, we obtain
    \begin{align}\label{eq:risk_expanded}
        R(w) &= N \int \Exp[\|(\htheta_W - \btheta_l)\|^2 + (1-w(h_N))^2 a(h_N) - 2(1-w(h_N))b(h_N)] f_{H_N}(h_N) dh_N, 
    \end{align}
    where 
    \begin{align*}
        a(h_N) &= \Exp[\|\htheta_W - \htheta_S\|^2 \mid h_N], \\
        b(h_N) &= \Exp[(\htheta_W - \btheta_l)^\top(\htheta_W - \htheta_S )\mid h_N].
    \end{align*}
    Since $\Exp[\|(\htheta_W - \btheta_l)\|^2]$ does not depend on $w$, minimizing Eq.~\eqref{eq:risk_expanded} over $w$ reduces to a pointwise optimization for each $h_N$ and we only have to consider the second and third component. 

    For each fixed $h_N > 0$, the integrand in Eq.~\eqref{eq:risk_expanded} is quadratic in $w(h_N)$ with positive leading coefficient $a(h_N)$. Minimizing over $w(h_N)$ yields the oracle pointwise optimal weight
    \begin{equation*}
        w^*(h_N) = 1-\frac{b(h_N)}{a(h_N)}.
    \end{equation*}
    By appling such optimal weight to the excess risk of any weight function $\hat{w}^*$ relative to $w^*$, the risk function satisfies
    \begin{align}\label{eq:excess_risk}
        R(\hat{w}^*) - R(w^*) = N \int a(h_N) [\hat{w}^*(h_N) - w^*(h_N)]^2 f_{H_N}(h_N) dh_N.
    \end{align}
    Since $a(h_N) > 0$ and the integrand is a product of non-negative terms, the right-hand side of Eq.~\eqref{eq:excess_risk} is non-negative for any weight function $\hat{w}^*$, confirming that $w^*$ defined in Eq.~\eqref{wstar} is indeed the pointwise risk minimum point.
    
    We relate $a(h_N)$ to $h_N$ through the quadratic form structure.
    We recall that $D_\delta = \hV_H^{-1/2}\sqrt{N}(\htheta_W - \htheta_S)$, so that $H_N = \|D_\delta\|^2$.
    Therefore,
    $a(h_N) = \Exp[\|\htheta_W - \htheta_S\|^2 \mid h_N]$ can be controled through the value of $h_N$ since
    \begin{equation*}
        N\|\htheta_W - \htheta_S\|^2 = D_\delta^\top \hV_H D_\delta \leq \lambda_{max}(\hV_H) \|D_\delta\|^2 = \lambda_{max}(\hV_H) H_N.
    \end{equation*}
    Taking conditional expectations on both sides, we obtain
    \begin{align}\label{eq:a_bound}
        N \cdot a(h_N) \leq \lambda_{\max}(\hV_H) h_N \sim \left(\lambda_{\max}(V_H) + \op\right) h_N,
    \end{align}
    since $\hV_H \cvp V_H$ (as established in the plug-in consistency results).

    We next prove that $[\hat{w}^*(h_N) - w^*(h_N)]^2 =\Op$.
    We first decompose the plug-in parameter as $\hat{\Theta} = (\hat{\Theta}_{-\lambda},  \hat\lambda)$, where $\hat{\Theta}_{-\lambda} = (\Sigma_W, \hat{C}_{WS}, \hV_H)$ collects the variance components and $\hat\lambda = H_N$.
    %By the lemmas~\ref{asym_AW} and \ref{asym_BW}, each variance component satisfies $\hat{\Theta}_{-\lambda} - \Theta_{-\lambda} = O_{P}(N^{-1/2})$.
    %Since $w^*(h_N;\cdot,\lambda)$ is a smooth function of $\Theta_{-\lambda}$ (the ratio $b(h_N)/a(h_N)$ depends smoothly on $V_W$, $C_{WS}$, and $V_H$), a Taylor expansion yields
    % \begin{equation*}
    %     w^*(h_N; \hat{\Theta}_{-\lambda}, \lambda) - w^*(h_N; \Theta_{-\lambda}, \lambda) = O_{P}(N^{-1/2})
    %     \quad \text{for each fixed } h_N.
    % \end{equation*}

    The estimator $\hat\lambda = H_N$ satisfies $\hat\lambda \xrightarrow{d} \chi^2_p(\lambda)$, where $\lambda = \delta_s^\top V_H^{-1} \delta_s$. 
    Crucially, $\hat\lambda$ converges in distribution to a non-degenerate random variable rather than in probability to the constant $\lambda$. 
    Consequently, for each fixed $h_N$,
    \begin{equation*}
         w^*(h_N;\hat\lambda) - w^*(h_N;\lambda) = \Op,
    \end{equation*}
    and this error does not vanish as $N \to \infty$.
    The $O_{P}(N^{-1/2})$ contribution from $\htheta_{-\lambda}$ is asymptotically negligible relative to the $\Op$ contribution from $\hat\lambda$, thus the overall weight estimation error satisfies
    \begin{equation}\label{eq:weight_error}
        [\hat{w}^*(h_N) - w^*(h_N)]^2 = \Op \quad \text{uniformly in } h_N.
    \end{equation}
    Moreover, since $w^* \in [0,1]$, the squared error is deterministically bounded by a constant say $c_w$ $\Op$.

    Substituting Eq.~\eqref{eq:a_bound} and Eq.~\eqref{eq:weight_error} into the excess risk identity Eq.~\eqref{eq:excess_risk}, we obtain
    \begin{align}\label{eq:final_bound}
        R(\hat{w}^*) - R(w^* )
        &= N \int a(h_N) [\hat{w}^*(h_N) - w^*(h_N)]^2 f_{H_N}(h_N) dh_N \notag \\
        &\leq \lambda_{\max}(V_H) \int [\hat{w}^*(h_N) - w^*(h_N)]^2  h_N  f_{H_N}(h_N)  dh_N + \op.
    \end{align}
    Since $[\hat{w}^*(h_N) - w^*(h_N)]^2 \leq c_w$ and $H_N \xrightarrow{d} \chi^2_p(\lambda)$, we have
    \begin{equation*}
        \int [\hat{w}^*(h_N) - w^*(h_N)]^2  h_N  f_{H_N}(h_N)  dh_N
        \leq c_w \int_0^\infty h_N  f_{H_N}(h_N) dh_N
        = c_w  \Exp[H_N].
    \end{equation*}
    For any $\chi^2_p(\lambda)$ random variable, $\Exp[\chi^2_p(\lambda)] = p + \lambda < \infty$.
    Therefore,
    \begin{equation}\label{eq:final_result}
        R(\hat{w}^*) - R(w^*) \leq c_w \lambda_{\max}(V_H)(p + \lambda) + \op = \Op.
    \end{equation}
    This establishes that the excess risk is bounded but does not vanish, due to the irreducible estimation error in $\hat\lambda$.
\end{proof}

\subsection{A constant w}

According to lemma~\ref{optimal_w_HN}, we also can find that, if we ignore the influence of $H_N$ in the weight $w$, i.e. we treat $w$ as a constant, then the ture risk function turns to 
\begin{equation*}
    \lN R(w) = \lN \Exp[tr( [w(\htheta_W - \htheta_l) + (1-w)(\htheta_S - \htheta_l)] [w(\htheta_W - \htheta_l) + (1-w)(\htheta_S - \htheta_l)]^\top )],
\end{equation*}
where $w$ is irrelevant with the estimators $\htheta_W$ and $\htheta_S$.
Then 
\begin{align}
    \lN R(w) = w^2 tr(\Sigma_W^{-1}) +2w(1-w) tr(C_{WS}) + (1-w)^2(tr(\Sigma_{SM}^{-1}) \|\delta_s\|^2).
\end{align}
Then the optimal oracle weight $w^*_{or}$ is only related with $\delta_s$ and
\begin{align}
    w^*_{or}(\delta_s) &= \arg \min_{w} \lN R(w) = 1 - \frac{\|\delta_s\|^2 + tr(\Sigma_{SM}^{-1} - C_{WS})}{\|\delta_s\|^2 + tr(\Sigma_{W}^{-1} + \Sigma_{SM}^{-1} - 2C_{WS})}.
\end{align}
Since $\|\delta_s\|$ is unknown, we also provide the minimax result of $w$ which is
\begin{align}
    w^*_{or,max} &= \arg \min_{w} \max_{\|\theta_W - \theta_S\| \leq M/\sqrt{N}} \lN R(w) = 1- \frac{\|M\|^2 + tr(\Sigma_{SM}^{-1} - C_{WS})}{\|M\|^2 + tr(\Sigma_{W}^{-1} + \Sigma_{SM}^{-1} - 2C_{WS})},
\end{align}
which means $w^*_{or,max}$ is the minimax optimal value of the MSE under the condition of $\|\theta_W - \theta_S\| \leq \frac{M}{\sqrt{N}}$.
